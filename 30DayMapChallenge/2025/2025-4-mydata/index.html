Read files and combine

# filenames
filene <-  list.files(here("data/strava"), pattern = "\\.gpx$", full.names = TRUE)


# read files and combine them
sporene <- lapply(filene, function(fil) {
  # read track points
  spor <- st_read(fil, layer = "track_points", quiet = TRUE)
  # add file names
  spor$fil <- basename(fil)
  spor
})

# combine to a sf object
spor_sf <- do.call(rbind, sporene)

Calculate total

# Calculate km for the displayed routes
bbox <- st_bbox(c(xmin = 5.0, ymin = 60.2, xmax = 5.35, ymax = 60.41), crs = st_crs(4326))

# lines displayed in the map
spor_i_kart <-  st_crop(spor_sf, bbox)

# combine points to lines
spor_linjer <- spor_i_kart |>
  group_by(fil) |>
  summarise(do_union = FALSE) |>
  st_cast("LINESTRING")

# calculate total length in km
total_km <- sum(st_length(spor_linjer)) / 1000

Add roads

Trying OSM road for the first time.

# Get roads from OSM
roads <- opq(bbox = bbox) |>
  add_osm_feature(key = "highway", 
                  value = c("motorway", "trunk", "primary", "secondary", "tertiary")) |>
  osmdata_sf()

# Extract the lines
roads_lines <- roads$osm_lines
